Dynamically Loading Content with JavaScript

A simple guide to creating a loader that dynamically loads content into your webpage from files. This displays a brief of your content as a blog tile.

Code Walkthrough

Let's break down the script piece by piece so you understand exactly what's happening.

Step 1: Configuration

const post_FOLDER = 'posts';
const MAX_POSTS = 50;

const container = document.getElementById('postsContainer');
  • post_FOLDER — The folder where your posts are stored. Change this if you use a different name.
  • MAX_POSTS — How many post folders to scan (post1 through post50). Bump this up if you have more.
  • container — Grabs the HTML element where posts will be displayed. Must have id="postsContainer".

Step 2: Load a Single Post

function loadPostFromFolder(folderPath) {
    const fileUrl = folderPath + '/post_index.html';
    
    return fetch(fileUrl)
        .then(function(response) {
            if (!response.ok) return null;
            return response.text();
        })
        .then(function(htmlText) {
            if (!htmlText) return null;
            
            const parser = new DOMParser();
            const document = parser.parseFromString(htmlText, 'text/html');
            
            function findMeta(name) {
                const metaTag = document.querySelector('meta[name="' + name + '"]');
                if (metaTag) {
                    return metaTag.getAttribute('content') || '';
                }
                return '';
            }
            
            const post = {
                title: findMeta('title') || 'Untitled',
                description: findMeta('description') || '',
                author: findMeta('author') || '',
                date: findMeta('date') || '',
                tags: findMeta('keywords') || '',
                folder: folderPath
            };
            
            // Clean up tags
            const tagsArray = post.tags.split(',');
            const cleanedTags = [];
            for (let i = 0; i < tagsArray.length; i++) {
                const cleanTag = tagsArray[i].trim();
                if (cleanTag.length > 0) {
                    cleanedTags.push(cleanTag);
                }
            }
            post.tags = cleanedTags;
            
            return post;
        })
        .catch(function(error) {
            return null;
        });
}
  • Fetch — Requests the post_index.html file from the folder.
  • Check response — If the file doesn't exist (404), returns null.
  • DOMParser — Parses the HTML so we can search for meta tags.
  • findMeta() — Helper function that looks for a meta tag by name and returns its content.
  • Post object — Gathers all metadata into a clean object: title, description, author, date, tags, and folder.
  • Tag cleaning — Splits comma-separated tags, trims whitespace, and removes empty entries.
  • Error handling — If anything fails, returns null so the script can skip that folder.

Step 3: Load All Posts

function loadAllPosts() {
    const loadingTasks = [];
    
    for (let i = 1; i <= MAX_POSTS; i++) {
        const folderPath = post_FOLDER + '/post' + i;
        const task = loadPostFromFolder(folderPath)
            .then(function(post) {
                if (post) {
                    post.index = i;
                    return post;
                }
                return null;
            });
        loadingTasks.push(task);
    }
    
    return Promise.all(loadingTasks)
        .then(function(allPosts) {
            const validPosts = [];
            for (let i = 0; i < allPosts.length; i++) {
                if (allPosts[i] !== null) {
                    validPosts.push(allPosts[i]);
                }
            }
            validPosts.sort(function(a, b) {
                return b.index - a.index;
            });
            return validPosts;
        });
}
  • Loop — Iterates from 1 to MAX_POSTS (50 by default), creating paths like posts/post1/, posts/post2/, etc.
  • Promise array — Each loadPostFromFolder() call returns a promise; we store all of them in loadingTasks.
  • Promise.all() — Waits for ALL folders to finish loading (or fail) before continuing.
  • Filter nulls — Removes any folders that didn't contain a valid post.
  • Sort — Orders posts by index number, highest first (so newer posts appear at the top).

Step 4: Display Posts

function displayPosts(posts) {
    container.innerHTML = '';
    
    if (posts.length === 0) {
        container.innerHTML = '<p>No posts found. Create posts/post1/post_index.html etc.</p>';
        return;
    }
    
    for (let i = 0; i < posts.length; i++) {
        const post = posts[i];
        
        const postDiv = document.createElement('div');
        postDiv.style.marginBottom = '1.8rem';
        postDiv.style.borderBottom = '1px solid #ccc';
        postDiv.style.paddingBottom = '0.8rem';
        
        // Title (linked to the full post)
        const titleLink = document.createElement('a');
        titleLink.href = post.folder + '/post_index.html';
        titleLink.textContent = post.title;
        const titleHeading = document.createElement('h3');
        titleHeading.style.margin = '0 0 0.2rem 0';
        titleHeading.appendChild(titleLink);
        postDiv.appendChild(titleHeading);
        
        // Metadata: Author, Date, Tags
        const metaParagraph = document.createElement('p');
        metaParagraph.style.margin = '0.2rem 0';
        let metaText = '';
        if (post.author) metaText += 'Author: ' + post.author;
        if (post.date) {
            if (metaText) metaText += '  ·  ';
            metaText += 'Posted: ' + post.date;
        }
        if (post.tags.length > 0) {
            if (metaText) metaText += '  ·  ';
            metaText += 'Tags: ' + post.tags.join(', ');
        }
        metaParagraph.textContent = metaText || 'No metadata';
        postDiv.appendChild(metaParagraph);
        
        // Description
        if (post.description) {
            const descriptionParagraph = document.createElement('p');
            descriptionParagraph.textContent = post.description;
            descriptionParagraph.style.margin = '0.4rem 0 0 0';
            postDiv.appendChild(descriptionParagraph);
        }
        
        container.appendChild(postDiv);
    }
}
  • Clear container — Wipes out any existing content.
  • Empty check — Shows a friendly message if no posts were found.
  • Loop through posts — Creates a <div> for each post.
  • Title — Wrapped in an <a> tag linking to the full post page.
  • Metadata — Displays author, date, and tags on one line, separated by ·.
  • Description — Shows the description as a short preview paragraph.
  • Inline styles — Adds basic spacing and borders for readability.

Step 5: Kick It Off

loadAllPosts()
    .then(function(posts) {
        displayPosts(posts);
    })
    .catch(function(error) {
        container.innerHTML = '<p>Error: ' + error.message + '</p>';
    });
  • Call loadAllPosts() — Starts the entire process.
  • .then() — Once all posts are loaded, passes them to displayPosts().
  • .catch() — If something goes wrong (network error, etc.), shows an error message.

How to Use This Script

  1. Create a folder called posts in your site's root directory.
  2. Inside posts, create numbered folders: post1, post2, post3, etc.
  3. In each folder, create a file named post_index.html.
  4. Add these meta tags to the <head> of each post_index.html:
<meta name="title" content="My Post Title">
<meta name="description" content="A short description.">
<meta name="author" content="Your Name">
<meta name="date" content="August 30, 2026">
<meta name="keywords" content="JavaScript, Web, GitHub">
  1. Add <div id="postsContainer"></div> to the page where you want the list to appear.
  2. Paste the script on the same page (or in an external .js file).

Follow Up

Now that you have a working dynamic loader, you can start creating posts! Just create a new folder (e.g., post3) and add a post_index.html file with the appropriate meta tags. The loader will automatically pick it up and display it on the main page.

While the loader is nice, also having the important data labeled as meta data if great (where have I heard that name before). This allows for easy searching and indexing of your posts, and can also be used for SEO purposes if you ever decide to make your blog public. Now Google and other tools can find your content for scraping too, which is a good thing if you want to share your work with the world. This is a bad thing if you are concerned about you blog being used for nefarious purposes, which in that case I recommend using tinfoil neatly wrapped around your computer. (But in all seriousness, if you are concerned about privacy, keep this on a local network or use a private server.)
